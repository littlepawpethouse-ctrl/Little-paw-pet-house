function updateModal(){ if(!currentKittens.length) return; const k = currentKittens[currentIndex]; modalImg.src = k.img || ''; modalName.textContent = k.name || ''; modalGender.textContent = 'Gender: ' + (k.gender || '-'); // Handle price and discount const priceText = k.price || ''; if(!priceText.toLowerCase().includes('sold')){ const priceNum = parseInt(priceText.replace('RM','').replace(/\D/g,'')); if(!isNaN(priceNum)){ const discounted = Math.round(priceNum * 0.8); modalOriginalPrice.textContent = 'RM' + priceNum; modalOriginalPrice.classList.remove('hidden'); modalPrice.textContent = 'RM' + discounted; modalDiscount.innerHTML = currentLang === 'zh' ? '🎉 年末特别优惠：八折优惠！' : currentLang === 'ms' ? '🎉 Diskaun 20% Akhir Tahun!' : '🎉 Year-End Special: 20% OFF!'; modalDiscount.classList.remove('hidden'); } else { modalOriginalPrice.classList.add('hidden'); modalPrice.textContent = priceText; modalDiscount.classList.add('hidden'); } } else { // Sold → hide discount modalOriginalPrice.classList.add('hidden'); modalPrice.textContent = priceText; modalDiscount.classList.add('hidden'); } // Multi-language description if(currentLang === 'en') modalDesc.textContent = k.descEn || k.desc || ''; else if(currentLang === 'zh') modalDesc.textContent = k.descZh || k.desc || ''; else if(currentLang === 'ms') modalDesc.textContent = k.descMs || k.desc || ''; else modalDesc.textContent = k.desc || ''; // Disable prev/next if only one kitten prevBtn.disabled = nextBtn.disabled = currentKittens.length <= 1; }
